// 改成你自己的接口地址 const CREATE_CODE_API = "https://rnrppmyrvauogudyjekq.supabase.co/functions/v1/createcode"; // 存储 key const STORAGE_WIN = "ai_win_count"; const STORAGE_REWARD = "ai_available_reward"; const STORAGE_CODES = "ai_gift_codes"; function getAiWinData() { return { winCount: Number(localStorage.getItem(STORAGE_WIN) || "0"), availableReward: Number(localStorage.getItem(STORAGE_REWARD) || "0") }; } function saveAiWinData(d) { localStorage.setItem(STORAGE_WIN, String(d.winCount)); localStorage.setItem(STORAGE_REWARD, String(d.availableReward)); } function getLocalAiCodes() { try { return JSON.parse(localStorage.getItem(STORAGE_CODES) || "[]"); } catch { return []; } } function saveLocalAiCodes(arr) { localStorage.setItem(STORAGE_CODES, JSON.stringify(arr)); } function copyCode(code, row, btn) { function restore() { row.classList.remove("copied"); btn.textContent = "复制"; } function done() { row.classList.add("copied"); btn.textContent = "已复制 ✓"; setTimeout(restore, 1500); } function fallback() { const ta = document.createElement("textarea"); ta.value = code; ta.style.position = "fixed"; ta.style.opacity = "0"; document.body.appendChild(ta); ta.select(); try { document.execCommand("copy"); } catch (e) {} document.body.removeChild(ta); done(); } if (navigator.clipboard && navigator.clipboard.writeText) { navigator.clipboard.writeText(code).then(done).catch(fallback); } else { fallback(); } } function renderCodes() { const codes = getLocalAiCodes(); const listEl = document.getElementById("codeList"); listEl.innerHTML = ""; if (!codes.length) return; codes.slice().reverse().forEach(function (item) { const row = document.createElement("div"); row.className = "win-code-item"; const meta = document.createElement("div"); meta.className = "meta"; const codeEl = document.createElement("span"); codeEl.className = "code"; codeEl.textContent = item.code; const descEl = document.createElement("span"); descEl.className = "desc"; descEl.textContent = "人机胜场礼包"; meta.appendChild(codeEl); meta.appendChild(descEl); const reward = document.createElement("div"); reward.className = "reward"; reward.textContent = "+" + item.points + " 积分"; const copyBtn = document.createElement("button"); copyBtn.className = "win-code-copy"; copyBtn.type = "button"; copyBtn.textContent = "复制"; function doCopy() { copyCode(item.code, row, copyBtn); } row.addEventListener("click", doCopy); copyBtn.addEventListener("click", function (e) { e.stopPropagation(); doCopy(); }); row.appendChild(meta); row.appendChild(reward); row.appendChild(copyBtn); listEl.appendChild(row); }); } function renderAiRewardPanel() { const data = getAiWinData(); const progressText = document.getElementById("aiProgressText"); const progressBar = document.getElementById("aiProgressBar"); const availText = document.getElementById("aiAvailText"); const btn = document.getElementById("claimAiCodeBtn"); progressText.textContent = `人机对战胜利进度：${data.winCount % 10}/10`; progressBar.style.width = `${((data.winCount % 10) / 10) * 100}%`; availText.textContent = `可领取档位：${data.availableReward} 个（每赢10局可领1个60-70积分礼品码）`; btn.disabled = data.availableReward <= 0; renderCodes(); } const claimBtn = document.getElementById("claimAiCodeBtn"); const msgBox = document.getElementById("aiMsg"); claimBtn.onclick = async function () { msgBox.className = "ai-msg"; msgBox.textContent = "正在生成礼品码..."; claimBtn.disabled = true; const data = getAiWinData(); try { const rewardPoints = Math.floor(Math.random() * 11) + 60; const res = await fetch(CREATE_CODE_API, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ points: rewardPoints }) }); const ret = await res.json(); if (!ret.success) { msgBox.className = "ai-msg error"; msgBox.textContent = "生成失败，请稍后再试"; claimBtn.disabled = false; renderAiRewardPanel(); return; } data.availableReward -= 1; saveAiWinData(data); const localList = getLocalAiCodes(); localList.push({ code: ret.code, points: ret.points, time: Date.now() }); saveLocalAiCodes(localList); msgBox.className = "ai-msg success"; msgBox.textContent = `✅ 礼品码生成成功：${ret.code}`; } catch (err) { msgBox.className = "ai-msg error"; msgBox.textContent = "网络异常，请检查连接后重试"; claimBtn.disabled = false; } renderAiRewardPanel(); }; // 这个函数你要在游戏胜利的地方调用：赢一局就执行一次 function aiWinAddOne() { const d = getAiWinData(); d.winCount += 1; if (d.winCount % 10 === 0) { d.availableReward += 1; } saveAiWinData(d); renderAiRewardPanel(); } renderAiRewardPanel();
